
import warnings
warnings.filterwarnings("ignore")

import yfinance as yf
import pandas as pd
from backtesting import Backtest, Strategy
from IPython.display import display

# ---------------- Settings ----------------
TICKERS = ["RELIANCE", "TCS", "INFY", "HDFCBANK", "ICICIBANK"]
START = "2021-01-01"
END = "2026-09-30"
CASH = 200_000
# ------------------------------------------


# MACD Line
def MACD(close):
    close = pd.Series(close)
    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    return ema12 - ema26


# MACD Signal Line
def MACD_Signal(close):
    return MACD(close).ewm(span=9, adjust=False).mean()


class MACDStrategy(Strategy):

    stop = 0.07

    def init(self):
        self.macd = self.I(MACD, self.data.Close, name="MACD")
        self.signal = self.I(MACD_Signal, self.data.Close, name="Signal")

    def next(self):

        price = self.data.Close[-1]

        # MACD BUY SIGNAL
        if (
            not self.position
            and self.macd[-2] <= self.signal[-2]
            and self.macd[-1] > self.signal[-1]
        ):
            self.buy(sl=price * (1 - self.stop))

        # MACD SELL SIGNAL
        elif (
            self.position
            and self.macd[-2] >= self.signal[-2]
            and self.macd[-1] < self.signal[-1]
        ):
            self.position.close()


# ---------------- Run Backtesting ----------------

for t in TICKERS:

    df = yf.Ticker(f"{t}.NS").history(
        start=START,
        end=END,
        auto_adjust=True
    )

    if df.empty:
        print(f"No data for {t}")
        continue

    df.index = df.index.tz_localize(None)

    bt = Backtest(
        df,
        MACDStrategy,
        cash=CASH,
        commission=0.0012
    )

    stats = bt.run()

    print(f"\n===== {t} =====")

    print(
        f"Return {stats['Return [%]']:.1f}% | "
        f"Buy & Hold {stats['Buy & Hold Return [%]']:.1f}% | "
        f"Trades {stats['# Trades']} | "
        f"Win Rate {stats['Win Rate [%]']:.1f}% | "
        f"Max DD {stats['Max. Drawdown [%]']:.1f}%"
    )

    bt.plot(
        resample=False,
        superimpose=False,
        plot_return=False,
        plot_drawdown=False
    )

    tr = stats["_trades"]

    if tr.empty:
        print("No trades")
        continue

    table = pd.DataFrame({

        "Entry Date": tr["EntryTime"].dt.date,

        "Entry Price": tr["EntryPrice"].round(2),

        "Exit Date": tr["ExitTime"].dt.date,

        "Exit Price": tr["ExitPrice"].round(2),

        "Shares": tr["Size"],

        "Return %": (tr["ReturnPct"] * 100).round(2),

        "P&L ₹": tr["PnL"].round(0),

        "Days Held": tr["Duration"].dt.days,

    })

    display(table.style.map(
        lambda v: "color: green" if v > 0 else "color: red",
        subset=["Return %", "P&L ₹"]
    ))

import warnings
warnings.filterwarnings("ignore")

import yfinance as yf
import pandas as pd
from backtesting import Backtest, Strategy
from IPython.display import display

# ---------------- Settings ----------------
TICKERS = ["RELIANCE", "TCS", "INFY", "HDFCBANK", "ICICIBANK"]
START = "2021-01-01"
END = "2026-09-30"
CASH = 200_000
# ------------------------------------------


# Average Volume
def Volume_Avg(volume, n=20):
    return pd.Series(volume).rolling(n).mean()


class VolumeStrategy(Strategy):

    n = 20
    multiplier = 1.5
    stop = 0.07

    def init(self):
        self.vol_avg = self.I(
            Volume_Avg, self.data.Volume, self.n, name="Average Volume"
        )

    def next(self):

        price = self.data.Close[-1]

        # BUY CONDITION: High Volume + Price Increase
        if (
            not self.position
            and self.data.Volume[-1] > self.vol_avg[-1] * self.multiplier
            and self.data.Close[-1] > self.data.Open[-1]
        ):
            self.buy(sl=price * (1 - self.stop))

        # SELL CONDITION: Low Volume or Price Decrease
        elif (
            self.position
            and self.data.Volume[-1] < self.vol_avg[-1]
            and self.data.Close[-1] < self.data.Open[-1]
        ):
            self.position.close()


# ---------------- Run Backtesting ----------------

for t in TICKERS:

    df = yf.Ticker(f"{t}.NS").history(
        start=START, end=END, auto_adjust=True
    )

    if df.empty:
        print(f"No data for {t}")
        continue

    df.index = df.index.tz_localize(None)

    bt = Backtest(
        df, VolumeStrategy, cash=CASH, commission=0.0012
    )

    stats = bt.run()

    print(f"\n===== {t} =====")

    print(
        f"Return {stats['Return [%]']:.1f}% | "
